---------------- FILE: dut/SampleHold.va ----------------
`include "constants.vams"
`include "disciplines.vams"

module SampleHold(clk, vin, vout);

    input clk, vin;
    output vout;

    parameter real vtrans = 0.5;
    parameter real delay = 0;
    parameter real trans_time = 1p;
    parameter real clk_threshold = 0.9;

    electrical vout, vin, clk;

    real v;

    analog begin
        @(cross(V(clk) - clk_threshold, +1))
            v = V(vin);

        V(vout) <+ transition(v, delay, trans_time);
    end

endmodule
---------------- FILE: tb/bridges/sample_hold_bridge.vams ----------------
`include "disciplines.vams"

module sample_hold_bridge(vin_r, vin_a);

    input  wreal      vin_r;
    output electrical vin_a;

    analog begin
        V(vin_a) <+ vin_r;
    end

endmodule

module ms_sample_hold_top;

    real clk_drv;
    real vin_drv;

    wreal clk;
    wreal vin;
    wreal vout;

    assign clk = clk_drv;
    assign vin = vin_drv;

    SampleHold dut (
        .clk  (clk),
        .vin  (vin),
        .vout (vout)
    );

    // Clock: 0 V / 1.8 V
    initial begin

            virtual gp_adc_interface.driver
